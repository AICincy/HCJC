"""Lab-only HTTP gzip server for before/after artifacts, not production infrastructure."""
import functools
import gzip
import io
import threading
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

@functools.lru_cache(maxsize=4096)
def payload(path, compress):
    data = Path(path).read_bytes()
    return gzip.compress(data, mtime=0) if compress else data

class Handler(SimpleHTTPRequestHandler):
    def send_head(self):
        path = Path(self.translate_path(self.path))
        if path.is_dir():
            path /= 'index.html'
        if not path.is_file():
            self.send_error(404)
            return None
        typ = self.guess_type(str(path))
        compress = ('gzip' in self.headers.get('Accept-Encoding', '') and
                    (typ.startswith('text/') or typ in ('application/javascript', 'application/json')))
        body = payload(str(path), compress)
        self.send_response(200)
        self.send_header('Content-Type', typ)
        self.send_header('Content-Length', str(len(body)))
        self.send_header('Cache-Control', 'no-store')
        if compress:
            self.send_header('Content-Encoding', 'gzip')
        self.end_headers()
        return io.BytesIO(body)

servers = [ThreadingHTTPServer(('0.0.0.0', port), functools.partial(Handler, directory=root))
           for port, root in [(8778, '.cache/closure/baseline-site'), (8779, '.cache/closure/site')]]
threading.Thread(target=servers[0].serve_forever, daemon=True).start()
servers[1].serve_forever()
